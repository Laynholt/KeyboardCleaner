cmake_minimum_required(VERSION 3.24)
project(KeyboardCleaner VERSION 1.0.1 LANGUAGES CXX RC)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
configure_file(src/AppVersion.h.in generated/AppVersion.h @ONLY)
set_source_files_properties(resources/app.rc resources/updater.rc PROPERTIES
    OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/resources/update.ps1")

set(WCW_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../win32-custom-widgets" CACHE PATH "Win32 Custom Widgets source directory")
add_subdirectory("${WCW_SOURCE_DIR}" win32-custom-widgets)

add_library(KeyboardCleanerUpdate STATIC src/Update.cpp)
target_include_directories(KeyboardCleanerUpdate PUBLIC src "${CMAKE_CURRENT_BINARY_DIR}/generated" PRIVATE third_party)
target_compile_features(KeyboardCleanerUpdate PUBLIC cxx_std_20)
target_compile_definitions(KeyboardCleanerUpdate PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(KeyboardCleanerUpdate PUBLIC winhttp bcrypt)
if(MSVC)
    target_compile_options(KeyboardCleanerUpdate PRIVATE /W4 /utf-8)
endif()

add_executable(KeyboardCleaner WIN32 src/main.cpp src/KeyboardHook.cpp resources/app.rc)
target_link_libraries(KeyboardCleaner PRIVATE Win32CustomWidgets::Win32CustomWidgets KeyboardCleanerUpdate dwmapi wtsapi32)
target_compile_features(KeyboardCleaner PRIVATE cxx_std_20)
set_target_properties(KeyboardCleaner PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(MSVC)
    target_compile_options(KeyboardCleaner PRIVATE /W4 /utf-8)
    target_link_options(KeyboardCleaner PRIVATE /MANIFEST:NO)
endif()
add_custom_command(TARGET KeyboardCleaner POST_BUILD
    COMMAND ${CMAKE_COMMAND} "-DINPUT=$<TARGET_FILE:KeyboardCleaner>"
            "-DOUTPUT=$<TARGET_FILE_DIR:KeyboardCleaner>/SHA256SUMS.txt"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/scripts/write-checksum.cmake")
